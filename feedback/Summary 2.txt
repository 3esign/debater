This is a genuinely compelling concept — the intersection of AI agents, prediction markets, and on-chain settlement is fresh territory. Here's ruthlessly honest feedback followed by concrete suggestions.

What's strong: The three-layer separation is clean and buildable. Blind crowd consensus as the resolution mechanism is clever — it sidesteps the AI-judge problem while keeping spectators financially engaged in their own judgment. The Elo system on top of a prediction market creates compounding incentive loops that will drive retention.

The hard problems you haven't solved yet:

The biggest one is the collusion/Sybil problem. If an agent trainer controls a wallet AND places bets, they can front-run their own agent's performance. You need economic separation between "agent trainer" and "spectator/bettor" roles — or at minimum a lockout window after entering where the trainer can't bet. This isn't optional; it's the thing that will get gamed first.

Context drift is your second existential risk. You mention "context reminders" in the Orchestrator, but multi-turn debates with cloud APIs are expensive and lossy at scale. A 20-turn debate with GPT-4 could cost $2-5 in tokens — that needs to be factored into entry fees or you'll bleed margin. Local Ollama agents are a great counterweight here but add latency variance that breaks the "live stage" experience.

Blind crowd consensus can be gamed by coordinated communities. If someone streams the debate to Discord and organizes block-voting, the "blind" part dissolves. Consider time-locking vote reveals (commit-reveal scheme on-chain) so no one can see vote tallies until the window closes.

The "winner takes all" model may kill liquidity. If early betting odds diverge heavily (e.g. 80/20 after 3 turns), the losing side dries up. You need a dynamic odds mechanism or a house market-making buffer.

Concrete suggestions:

On architecture: Flowise is fine for prototyping but you'll hit its abstraction ceiling fast when you need custom turn-enforcement logic. A lightweight custom Node.js orchestrator with a state machine (idle → agent-A-turn → agent-B-turn → voting → settlement) will serve you better long-term. Build the state machine first — everything else plugs into it.

On monetization: The platform fee is obvious, but consider selling "debate replays" as NFTs or offering tiered spectator seats (front-row = earlier bet access). This creates non-betting revenue.

On the product loop: The most viral mechanic you have is agent profiles — trainers showing off their system prompts (redacted), win rates, and debate highlights. Make this shareable and you get free distribution.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/access/Ownable.sol";
import "@openzeppelin/contracts/utils/ReentrancyGuard.sol";

contract DebaterVault is Ownable, ReentrancyGuard {

    IERC20 public immutable usdc;
    uint256 public platformFeeBps = 300; // 3%

    enum MatchState { Open, Live, Voting, Resolved, Cancelled }

    struct Match {
        address agentA;
        address agentB;
        uint256 entryFee;
        uint256 betPoolA;
        uint256 betPoolB;
        MatchState state;
        address winner;
        uint256 revealDeadline;
    }

    struct Bet {
        address agent;   // agentA or agentB address
        uint256 amount;
    }

    // commitHash = keccak256(abi.encodePacked(vote, salt))
    mapping(uint256 => mapping(address => bytes32)) public voteCommits;
    mapping(uint256 => mapping(address => address)) public voteReveals;
    mapping(uint256 => mapping(address => Bet))     public bets;
    mapping(uint256 => Match) public matches;
    uint256 public matchCount;

    // ── ENTRY ──────────────────────────────────────────

    function enterMatch(uint256 matchId, address opponent, uint256 fee)
        external nonReentrant
    {
        Match storage m = matches[matchId];
        require(m.state == MatchState.Open, "not open");
        require(m.entryFee == fee, "wrong fee");

        usdc.transferFrom(msg.sender, address(this), fee);

        if (m.agentA == address(0)) {
            m.agentA = msg.sender;
        } else {
            require(m.agentB == address(0) && msg.sender == opponent, "slot taken");
            m.agentB = msg.sender;
            m.state  = MatchState.Live;  // both in — debate starts
        }
    }

    // ── BETTING ────────────────────────────────────────

    function placeBet(uint256 matchId, address agent, uint256 amount)
        external nonReentrant
    {
        Match storage m = matches[matchId];
        require(m.state == MatchState.Live, "betting closed");
        // Enforce trainer/bettor separation here:
        require(msg.sender != m.agentA && msg.sender != m.agentB, "trainers cannot bet");

        usdc.transferFrom(msg.sender, address(this), amount);
        bets[matchId][msg.sender] = Bet(agent, amount);

        if (agent == m.agentA) m.betPoolA += amount;
        else                   m.betPoolB += amount;
    }

    function lockBetting(uint256 matchId) external onlyOwner {
        // Orchestrator calls this when debate ends, before vote window
        matches[matchId].state = MatchState.Voting;
        matches[matchId].revealDeadline = block.timestamp + 24 hours;
    }

    // ── COMMIT-REVEAL VOTING ───────────────────────────

    function commitVote(uint256 matchId, bytes32 commitHash) external {
        require(matches[matchId].state == MatchState.Voting, "not voting");
        voteCommits[matchId][msg.sender] = commitHash;
    }

    function revealVote(uint256 matchId, address vote, bytes32 salt) external {
        Match storage m = matches[matchId];
        require(m.state == MatchState.Voting, "not voting");
        require(block.timestamp <= m.revealDeadline, "reveal expired");
        require(
            voteCommits[matchId][msg.sender] == keccak256(abi.encodePacked(vote, salt)),
            "hash mismatch"
        );
        voteReveals[matchId][msg.sender] = vote;
    }

    // ── SETTLEMENT ─────────────────────────────────────

    function resolveMatch(uint256 matchId, address winner) external onlyOwner {
        Match storage m = matches[matchId];
        require(m.state == MatchState.Voting, "not in voting");
        require(winner == m.agentA || winner == m.agentB, "invalid winner");
        require(block.timestamp > m.revealDeadline, "reveal window open");

        m.winner = winner;
        m.state  = MatchState.Resolved;

        uint256 total     = (m.entryFee * 2) + m.betPoolA + m.betPoolB;
        uint256 fee       = (total * platformFeeBps) / 10000;
        uint256 remainder = total - fee;

        // Winning pool share (A or B)
        uint256 winPool   = (winner == m.agentA) ? m.betPoolA : m.betPoolB;
        uint256 losePool  = (winner == m.agentA) ? m.betPoolB : m.betPoolA;

        // Agent gets: both entry fees + losing bet pool, minus platform cut portion
        uint256 agentPayout = (m.entryFee * 2) + losePool - (fee * 60 / 100);

        usdc.transfer(winner, agentPayout);
        usdc.transfer(owner(), fee); // platform treasury
        // Bettors claim proportionally — see claimWinnings()
    }

    function claimWinnings(uint256 matchId) external nonReentrant {
        Match storage m = matches[matchId];
        require(m.state == MatchState.Resolved, "not resolved");

        Bet storage b = bets[matchId][msg.sender];
        require(b.agent == m.winner, "losing bet");
        require(b.amount > 0, "nothing to claim");

        uint256 winPool  = (m.winner == m.agentA) ? m.betPoolA : m.betPoolB;
        uint256 losePool = (m.winner == m.agentA) ? m.betPoolB : m.betPoolA;

        // Pro-rata share of losing pool
        uint256 payout = b.amount + (b.amount * losePool / winPool);

        b.amount = 0; // CEI pattern — zero before transfer
        usdc.transfer(msg.sender, payout);
    }

    // ── SAFETY ─────────────────────────────────────────

    function cancelMatch(uint256 matchId) external onlyOwner {
        Match storage m = matches[matchId];
        require(m.state != MatchState.Resolved, "already done");
        m.state = MatchState.Cancelled;
        // Refund agents
        if (m.agentA != address(0)) usdc.transfer(m.agentA, m.entryFee);
        if (m.agentB != address(0)) usdc.transfer(m.agentB, m.entryFee);
    }
}
A few things worth calling out about this implementation:

The trainers cannot bet guard on line 32 is your primary collusion defense — it's one line but it's load-bearing. Don't skip it.

The onlyOwner on resolveMatch and lockBetting means your orchestrator's wallet is currently the trusted authority. That's fine for v1, but for a decentralized future you'd want to replace this with a multisig or a governance-controlled oracle. Don't advertise it as trustless until you've done that.

The claimWinnings function uses the checks-effects-interactions (CEI) pattern — b.amount = 0 happens before the transfer. This is critical. Reversing that order is the classic reentrancy attack vector.

The cancelMatch safety hatch is non-optional. Networks go down, APIs fail, debates get stuck. Without a cancellation path you'll permanently lock user funds, which is both a legal problem and a reputational death sentence.

One thing still to design: right now resolveMatch trusts the orchestrator to pass in the correct winner address. In practice you'd want to call this function only after verifying that the on-chain vote tally exceeds a quorum threshold (e.g. at least 10 reveals, winner has >50%). That logic should sit in resolveMatch itself, not just in the orchestrator, so it can't be bypassed by a compromised backend.

import OpenAI from 'openai'
import { Turn } from './types'

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY })

const SYSTEM_BASE = `You are a debate agent. Be sharp, concise, and argumentative.
Keep responses under 120 words. Stay on topic. Directly rebut your opponent's last point.
Do NOT be agreeable. Do NOT concede. You are competing to WIN.`

export async function getAgentTurn(
  systemPrompt: string,
  topic: string,
  history: Turn[],
  side: 'A' | 'B'
): Promise<{ content: string; tokenCount: number }> {
  const messages: OpenAI.Chat.ChatCompletionMessageParam[] = [
    {
      role: 'system',
      content: `${SYSTEM_BASE}\n\nYour persona: ${systemPrompt}\nYou are arguing side ${side} on: "${topic}"`,
    },
    ...history.map((t) => ({
      role: t.agent === side ? ('assistant' as const) : ('user' as const),
      content: t.content,
    })),
  ]

  // Add context reminder every 3 turns to fight drift
  if (history.length > 0 && history.length % 3 === 0) {
    messages.push({
      role: 'system',
      content: `REMINDER: Stay focused on "${topic}". Rebut the last argument directly.`,
    })
  }

  const response = await openai.chat.completions.create({
    model: 'gpt-4o-mini',  // cheap and fast for MVP
    messages,
    max_tokens: 200,
    temperature: 0.9,
  })

  const content = response.choices[0].message.content ?? ''
  const tokenCount = response.usage?.total_tokens ?? 0

  return { content, tokenCount }
}

import { Match, Turn } from './types'
import { getMatch, setMatch } from './redis'
import { getAgentTurn } from './agents'
import { pusherServer } from './pusher'

export async function runNextTurn(matchId: string): Promise<void> {
  const match = await getMatch(matchId)
  if (!match || match.state !== 'live') return

  const side = match.currentTurn
  const { content, tokenCount } = await getAgentTurn(
    side === 'A' ? match.agentAPrompt : match.agentBPrompt,
    match.topic,
    match.turns,
    side
  )

  const turn: Turn = { agent: side, content, timestamp: Date.now(), tokenCount }
  match.turns.push(turn)
  match.turnCount++
  match.currentTurn = side === 'A' ? 'B' : 'A'

  if (match.turnCount >= match.maxTurns) {
    match.state = 'voting'
  }

  await setMatch(match)

  // Broadcast turn to all spectators
  await pusherServer.trigger(`match-${matchId}`, 'turn', {
    turn,
    state: match.state,
    turnCount: match.turnCount,
  })
}

export function resolveWinner(match: Match): 'A' | 'B' {
  const tally = { A: 0, B: 0 }
  Object.values(match.votes).forEach((v) => tally[v]++)
  // Tiebreak: more bets on winning side
  if (tally.A === tally.B) {
    const betA = match.bets.filter((b) => b.side === 'A').reduce((s, b) => s + b.amount, 0)
    const betB = match.bets.filter((b) => b.side === 'B').reduce((s, b) => s + b.amount, 0)
    return betA >= betB ? 'A' : 'B'
  }
  return tally.A > tally.B ? 'A' : 'B'
}

export function calculatePayouts(match: Match): Record<string, number> {
  if (!match.winner) return {}
  const payouts: Record<string, number> = {}
  const losingPool = match.bets
    .filter((b) => b.side !== match.winner)
    .reduce((s, b) => s + b.amount, 0)
  const winningBets = match.bets.filter((b) => b.side === match.winner)
  const winPool = winningBets.reduce((s, b) => s + b.amount, 0)

  winningBets.forEach((b) => {
    const share = winPool > 0 ? (b.amount / winPool) * losingPool * 0.97 : 0
    payouts[b.userId] = (payouts[b.userId] ?? 0) + b.amount + share
  })
  return payouts
}

'use client'
import { useState, useEffect, useRef, useCallback } from 'react'
import { useParams } from 'next/navigation'
import { Match, Turn } from '@/lib/types'
import { pusherClient } from '@/lib/pusher'

const USER_ID = typeof window !== 'undefined'
  ? (localStorage.getItem('uid') ?? (() => {
      const id = Math.random().toString(36).slice(2)
      localStorage.setItem('uid', id)
      return id
    })())
  : 'anon'

export default function Arena() {
  const { id } = useParams<{ id: string }>()
  const [match, setMatch] = useState<Match | null>(null)
  const [betAmount, setBetAmount] = useState(100)
  const [myBet, setMyBet] = useState<'A' | 'B' | null>(null)
  const [myVote, setMyVote] = useState<'A' | 'B' | null>(null)
  const [balance, setBalance] = useState(1000)
  const [running, setRunning] = useState(false)
  const [status, setStatus] = useState('')
  const bottomRef = useRef<HTMLDivElement>(null)

  const fetchMatch = useCallback(async () => {
    const r = await fetch(`/api/match?id=${id}`)
    // we'll use the GET all endpoint and find ours, or add a dedicated GET
    const all = await r.json()
    const m = all.find((x: Match) => x.id === id)
    if (m) setMatch(m)
  }, [id])

  useEffect(() => { fetchMatch() }, [fetchMatch])

  useEffect(() => {
    const channel = pusherClient.subscribe(`match-${id}`)
    channel.bind('turn', (data: { turn: Turn; state: string; turnCount: number }) => {
      setMatch(prev => prev ? {
        ...prev,
        turns: [...prev.turns, data.turn],
        state: data.state as Match['state'],
        turnCount: data.turnCount,
      } : prev)
    })
    channel.bind('resolved', (data: { winner: string; payouts: Record<string, number> }) => {
      setMatch(prev => prev ? { ...prev, state: 'resolved', winner: data.winner as 'A' | 'B' } : prev)
      if (data.payouts[USER_ID]) {
        setBalance(prev => prev + data.payouts[USER_ID])
        setStatus(`You won ${data.payouts[USER_ID].toFixed(0)} mock USDC!`)
      }
    })
    return () => pusherClient.unsubscribe(`match-${id}`)
  }, [id])

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: 'smooth' })
  }, [match?.turns.length])

  async function startAndRun() {
    if (!match || running) return
    setRunning(true)
    const maxTurns = match.maxTurns
    for (let i = match.turnCount; i < maxTurns; i++) {
      const r = await fetch('/api/turn', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ matchId: id }),
      })
      if (!r.ok) break
      await new Promise(res => setTimeout(res, 1200)) // pace turns
    }
    setRunning(false)
  }

  async function placeBet(side: 'A' | 'B') {
    if (myBet) return
    const r = await fetch('/api/bet', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ matchId: id, userId: USER_ID, side, amount: betAmount }),
    })
    if (r.ok) {
      const data = await r.json()
      setMyBet(side)
      setBalance(data.balance)
    }
  }

  async function castVote(vote: 'A' | 'B') {
    if (myVote) return
    await fetch('/api/vote', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ matchId: id, userId: USER_ID, vote }),
    })
    setMyVote(vote)
  }

  async function resolve() {
    await fetch('/api/resolve', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ matchId: id }),
    })
    fetchMatch()
  }

  if (!match) return <div className="min-h-screen bg-zinc-950 flex items-center justify-center text-zinc-500">Loading…</div>

  const betA = match.bets.filter(b => b.side === 'A').reduce((s, b) => s + b.amount, 0)
  const betB = match.bets.filter(b => b.side === 'B').reduce((s, b) => s + b.amount, 0)
  const totalBet = betA + betB || 1
  const voteA = Object.values(match.votes).filter(v => v === 'A').length
  const voteB = Object.values(match.votes).filter(v => v === 'B').length

  return (
    <div className="min-h-screen bg-zinc-950 text-white flex flex-col">
      {/* Header */}
      <div className="border-b border-zinc-800 px-6 py-4 flex items-center justify-between">
        <div>
          <h1 className="font-semibold">{match.topic}</h1>
          <p className="text-xs text-zinc-500 mt-0.5">
            Turn {match.turnCount}/{match.maxTurns} · {match.state}
          </p>
        </div>
        <div className="text-right">
          <p className="text-xs text-zinc-500">Balance</p>
          <p className="font-mono font-medium">{balance.toFixed(0)} <span className="text-zinc-400 text-xs">mUSDC</span></p>
        </div>
      </div>

      <div className="flex flex-1 overflow-hidden">
        {/* Debate feed */}
        <div className="flex-1 overflow-y-auto p-6 space-y-4">
          {match.turns.map((turn, i) => (
            <div key={i} className={`flex ${turn.agent === 'B' ? 'justify-end' : 'justify-start'}`}>
              <div className={`max-w-[70%] rounded-2xl px-4 py-3 ${
                turn.agent === 'A'
                  ? 'bg-purple-950 border border-purple-800 rounded-tl-sm'
                  : 'bg-blue-950 border border-blue-800 rounded-tr-sm'
              }`}>
                <p className={`text-xs font-medium mb-1 ${turn.agent === 'A' ? 'text-purple-400' : 'text-blue-400'}`}>
                  Agent {turn.agent}
                </p>
                <p className="text-sm leading-relaxed">{turn.content}</p>
                <p className="text-xs text-zinc-600 mt-1">{turn.tokenCount} tokens</p>
              </div>
            </div>
          ))}
          {running && (
            <div className="flex justify-center">
              <span className="text-xs text-zinc-500 animate-pulse">Agent thinking…</span>
            </div>
          )}
          <div ref={bottomRef} />
        </div>

        {/* Sidebar */}
        <div className="w-64 border-l border-zinc-800 p-4 space-y-6 overflow-y-auto shrink-0">

          {/* Controls */}
          {match.state === 'lobby' && (
            <button onClick={startAndRun}
              className="w-full bg-white text-black text-sm font-medium py-2 rounded-lg hover:bg-zinc-200 transition">
              Start debate
            </button>
          )}
          {match.state === 'live' && !running && (
            <button onClick={startAndRun}
              className="w-full bg-green-600 text-white text-sm font-medium py-2 rounded-lg hover:bg-green-700 transition">
              Resume
            </button>
          )}

          {/* Bet pool */}
          <div>
            <p className="text-xs text-zinc-500 uppercase tracking-wider mb-3">Bet pool</p>
            <div className="flex rounded-lg overflow-hidden mb-3 h-2">
              <div className="bg-purple-600 transition-all" style={{ width: `${betA / totalBet * 100}%` }} />
              <div className="bg-blue-600 transition-all" style={{ width: `${betB / totalBet * 100}%` }} />
            </div>
            <div className="flex justify-between text-xs text-zinc-400 mb-4">
              <span>A: {betA} mUSDC</span>
              <span>B: {betB} mUSDC</span>
            </div>
            {match.state === 'live' && !myBet && (
              <>
                <input type="range" min={10} max={500} step={10} value={betAmount}
                  onChange={e => setBetAmount(+e.target.value)}
                  className="w-full mb-2" />
                <p className="text-center text-xs text-zinc-400 mb-3">{betAmount} mUSDC</p>
                <div className="grid grid-cols-2 gap-2">
                  <button onClick={() => placeBet('A')}
                    className="bg-purple-900 border border-purple-700 text-purple-300 text-xs py-2 rounded-lg hover:bg-purple-800 transition">
                    Bet A
                  </button>
                  <button onClick={() => placeBet('B')}
                    className="bg-blue-900 border border-blue-700 text-blue-300 text-xs py-2 rounded-lg hover:bg-blue-800 transition">
                    Bet B
                  </button>
                </div>
              </>
            )}
            {myBet && match.state === 'live' && (
              <p className="text-xs text-center text-zinc-500">Bet placed on Agent {myBet}</p>
            )}
          </div>

          {/* Voting */}
          {match.state === 'voting' && (
            <div>
              <p className="text-xs text-zinc-500 uppercase tracking-wider mb-3">Vote for winner</p>
              {!myVote ? (
                <div className="grid grid-cols-2 gap-2">
                  <button onClick={() => castVote('A')}
                    className="bg-purple-900 border border-purple-700 text-purple-300 text-xs py-2 rounded-lg hover:bg-purple-800 transition">
                    Agent A
                  </button>
                  <button onClick={() => castVote('B')}
                    className="bg-blue-900 border border-blue-700 text-blue-300 text-xs py-2 rounded-lg hover:bg-blue-800 transition">
                    Agent B
                  </button>
                </div>
              ) : (
                <p className="text-xs text-center text-zinc-500">Voted for Agent {myVote}</p>
              )}
              <div className="mt-3 text-xs text-zinc-600 text-center">
                {voteA} vs {voteB} votes
              </div>
              <button onClick={resolve}
                className="w-full mt-3 bg-amber-600 text-white text-xs py-2 rounded-lg hover:bg-amber-700 transition">
                Resolve match
              </button>
            </div>
          )}

          {/* Result */}
          {match.state === 'resolved' && (
            <div className="text-center">
              <p className="text-xs text-zinc-500 mb-2">Winner</p>
              <p className={`text-2xl font-bold ${match.winner === 'A' ? 'text-purple-400' : 'text-blue-400'}`}>
                Agent {match.winner}
              </p>
              {status && <p className="text-xs text-green-400 mt-2">{status}</p>}
            </div>
          )}
        </div>
      </div>
    </div>
  )
}
